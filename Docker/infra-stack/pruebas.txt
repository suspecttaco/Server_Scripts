PROTOCOLO DE PRUEBAS - infra-stack
===================================

PRUEBA 11.1 - Validacion de aislamiento de red

# Debe responder 200 con el HTML de la app
curl http://localhost:80

# Debe fallar o dar timeout (puerto no expuesto)
curl http://localhost:5432


PRUEBA 11.2 - Validacion de resolucion interna DNS

# Ejecutar ping al nombre del servicio desde el contenedor nginx
sudo docker exec infra-stack-nginx-1 ping -c 3 postgres


PRUEBA 11.3 - Validacion de tunel cifrado de gestion
-----------------------------------------------------
# Desde Windows PowerShell, abrir el tunel
# No requiere buscar la IP interna, pgadmin esta mapeado al loopback del host
ssh -L 8080:localhost:5050 usuario@IP_VM

# Abrir en el navegador
http://localhost:8080


PRUEBA 11.4 - Validacion de persistencia y healthcheck
-------------------------------------------------------
# Paso 1: crear datos en pgadmin antes de bajar el stack
# Abrir tunel (ver prueba 11.3) y ejecutar en Query Tool de pgadmin:
CREATE TABLE prueba (id SERIAL PRIMARY KEY, valor TEXT);
INSERT INTO prueba (valor) VALUES
    ('persistencia ok'),
    ('dato dos'),
    ('dato tres');

# Paso 2: desde la VM, bajar y volver a subir el stack
cd ~/infra-stack
sudo docker compose down
sudo docker compose up -d 2>&1

# Resultado esperado en el output: postgres aparece como Healthy
# antes de que pgadmin arranque

# Paso 3: abrir tunel y verificar datos en pgadmin
ssh -L 8080:localhost:5050 usuario@IP_VM

# Ejecutar en Query Tool de pgadmin:
SELECT * FROM prueba;

# Resultado esperado: los 3 registros insertados antes del reinicio


PRUEBA DE BALANCEO DE CARGA
----------------------------
# Desde la VM, con el stack corriendo

# Enviar 9 requests al balanceador
for i in {1..9}; do curl -s http://localhost:80 > /dev/null; done

# Verificar distribucion en los logs de nginx
sudo docker logs infra-stack-nginx-1 2>&1 | tail -9

# Resultado esperado: las peticiones se distribuyen en round-robin
# entre las 3 replicas de app (upstream=IP_1, IP_2, IP_3 rotando)
